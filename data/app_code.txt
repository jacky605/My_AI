def load_chat_history():
    if os.path.exists(HISTORY_FILE):
        try:
            with open(HISTORY_FILE, "r", encoding="utf-8") as f:
                return json.load(f)
        except Exception:
            pass
    return [{"role": "assistant", "content": "你好！我是混合 RAG 助手。你的對話紀錄會自動儲存！"}]

def save_chat_history(messages):
    try:
        with open(HISTORY_FILE, "w", encoding="utf-8") as f:
            json.dump(messages, f, ensure_ascii=False, indent=2)
    except Exception as e:
        st.error(f"儲存歷史紀錄失敗: {e}")

def fetch_context_by_query(prompt_text, vectorstore, web_search):
    """
    將檢索與 Fallback 邏輯抽離：
    1. 特定關鍵字 -> 讀取系統當前原始碼
    2. 向量相似度分數判定 -> 距離 < 0.6 採用本地知識庫 (避免二次 LLM 呼叫延遲)
    3. 否則 -> Fallback 聯網搜尋 (DuckDuckGo)
    """
    code_keywords = ["app.py", "系統原始碼", "系統程式碼", "系統架構", "app_code"]
    
    # 情況 A：查詢系統原始碼
    if any(kw in prompt_text.lower() for kw in code_keywords):
        st.info("💡 來源：系統當前原始碼 (app.py)")
        try:
            with open(CURRENT_APP_PATH, "r", encoding="utf-8") as f:
                return f"【完整系統原始碼 app.py】\n```python\n{f.read()}\n```"
        except Exception as e:
            return f"無法存取系統原始碼檔案：{e}"
            
    # 情況 B：標準向量檢索 (使用相似度距離分數)
    docs_and_scores = vectorstore.similarity_search_with_score(prompt_text, k=3)
    
    # L2/Cosine 距離越低代表越相似，門檻設為 0.6 (可依需求微調)
    if docs_and_scores and docs_and_scores[0][1] < 0.6:
        st.info("💡 來源：本地私有知識庫 (ChromaDB)")
        return "\n\n".join([doc.page_content for doc, _ in docs_and_scores])
        
    # 情況 C：Fallback 聯網搜尋
    st.warning("🌐 本地資料庫無相符記錄，自動升級為聯網搜尋 (DuckDuckGo)")
    try:
        return web_search.run(prompt_text)
    except Exception as e:
        return f"無法進行網路搜尋：{e}"

def convert_messages_for_langchain(messages_history):
    """轉換歷史對話格式，提供近 3 輪對話上下文 (前 6 則)"""
    langchain_history = []
    for m in messages_history[-7:-1]:
        if m["role"] == "user":
            langchain_history.append(HumanMessage(content=m["content"]))
        elif m["role"] == "assistant":
            langchain_history.append(AIMessage(content=m["content"]))
    return langchain_history

@st.cache_resource
def init_vectorstore():
    # 重新構建時先清空數據庫目錄，防止數據累積重複寫入
    if os.path.exists(CHROMA_DB_DIR):
        try:
            shutil.rmtree(CHROMA_DB_DIR)
        except Exception:
            pass

    embeddings = OllamaEmbeddings(model="nomic-embed-text")
    text_splitter = RecursiveCharacterTextSplitter(chunk_size=500, chunk_overlap=50)
    
    docs = []
    for filename in os.listdir(DATA_DIR):
        if filename.endswith(".txt"):
            filepath = os.path.join(DATA_DIR, filename)
            try:
                with open(filepath, "r", encoding="utf-8") as f:
                    text = f.read()
                    chunks = text_splitter.split_text(text)
                    for i, chunk in enumerate(chunks):
                        docs.append(f"【檔案: {filename} (區塊 {i+1})】\n{chunk}")
            except Exception as e:
                st.warning(f"讀取檔案 {filename} 失敗: {e}")

    if not docs:
        docs = ["目前本地知識庫無任何資料。"]

    return Chroma.from_texts(texts=docs, embedding=embeddings, persist_directory=CHROMA_DB_DIR)

@st.cache_resource
def init_models():
    return ChatOllama(model="olmo2", temperature=0.1), DuckDuckGoSearchRun()

vectorstore = init_vectorstore()
llm, web_search = init_models()

# Prompt 模組組裝
template = """你是一個專業的助手。請根據【參考資料】與【對話歷史】精準回答問題。

【參考資料】
{context}

注意：若資料包含程式碼，請直接分析，不要回答「無法讀取外部文件」。
"""

prompt = ChatPromptTemplate.from_messages([
    ("system", template),
    MessagesPlaceholder(variable_name="chat_history"),
    ("human", "{question}")
])
chain = prompt | llm | StrOutputParser()

# 初始化與載入對話歷程
if "messages" not in st.session_state:
    st.session_state.messages = load_chat_history()

for message in st.session_state.messages:
    with st.chat_message(message["role"]):
        st.markdown(message["content"])

# 處理使用者輸入
if prompt_text := st.chat_input("請輸入你的問題..."):
    st.session_state.messages.append({"role": "user", "content": prompt_text})
    save_chat_history(st.session_state.messages)
    
    with st.chat_message("user"):
        st.markdown(prompt_text)

    with st.chat_message("assistant"):
        with st.spinner("思考與檢索中..."):
            final_context = fetch_context_by_query(prompt_text, vectorstore, web_search)
            history = convert_messages_for_langchain(st.session_state.messages)
            
            # 使用 st.write_stream 與 chain.stream 實現即時打字機串流效果
            response_stream = chain.stream({
                "context": final_context, 
                "chat_history": history,
                "question": prompt_text
            })
            full_response = st.write_stream(response_stream)
            
    st.session_state.messages.append({"role": "assistant", "content": full_response})
    save_chat_history(st.session_state.messages)
